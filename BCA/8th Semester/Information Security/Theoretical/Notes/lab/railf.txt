#include<stdio.h>
#include<conio.h>
#include<string.h>
void main()
{
	int i,j,k,l;
	char a[20],c[20],d[20];
	clrscr();
	printf("\n\nEnter the input string : ");
	gets(a);
	l=strlen(a);
	/*Encrypting*/
	for(i=0,j=0;i<l;i++)
	{
		if(i%2==0)
			c[j++]=a[i];
	}
	for(i=0;i<l;i++)
	{
		if(i%2==1)
			c[j++]=a[i];
	}
	c[j]='\0';
	printf("\nEncrypted text:");
	printf("%s",c);
	/*Decrypting*/
	if(l%2==0)
		k=l/2;
	else
		k=(l/2)+1;
	for(i=0,j=0;i<k;i++)
	{
		d[j]=c[i];
		j=j+2;
	}
	for(i=k,j=1;i<l;i++)
	{
		d[j]=c[i];
		j=j+2;
	}
	d[l]='\0';
	printf("\nDecryption Text: ");
	printf("%s",d);
	getch();
}
